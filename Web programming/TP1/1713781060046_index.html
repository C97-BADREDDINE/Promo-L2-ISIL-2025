Exercice : Programme en C

Écrire un programme en langage C qui permet de :

  1. Remplir une matrice de taille 3x3 par des entiers saisis par l'utilisateur.
  2. Afficher la matrice.
  3. Calculer et afficher la somme des éléments positifs de la matrice.
            
     #include <stdio.h>
            
    // Définition de la taille de la matrice
    #define ROWS 3
    #define COLS 3
            
    // Fonction pour remplir la matrice
    void remplirMatrice(int matrice[ROWS][COLS]) {
    printf("Entrez les elements de la matrice %dx%d:\n", ROWS, COLS);
         for (int i = 0; i < ROWS; i++) {
              for (int j = 0; j < COLS; j++) {
                  printf("Element [%d][%d] : ", i + 1, j + 1);
                scanf("%d", &matrice[i][j]);
            }
        }
    }
            
    // Fonction pour calculer et afficher la somme des éléments positifs
    void sommePositifs(int matrice[ROWS][COLS]) {
    int somme = 0;
        for (int i = 0; i < ROWS; i++) {
                for (int j = 0; j < COLS; j++) {
                     if (matrice[i][j] > 0) {
                      somme += matrice[i][j];
                }
            }
        }
        printf("La somme des elements positifs de la matrice est : %d\n", somme);
    }
    
    // Fonction principale
    int main() {
        // Déclaration de la matrice
        int matrice[ROWS][COLS];
    
        // Appel de la fonction pour remplir la matrice
        remplirMatrice(matrice);
    
        // Affichage de la matrice
        printf("\nLa matrice que vous avez entre est :\n");
        for (int i = 0; i < ROWS; i++) {
            for (int j = 0; j < COLS; j++) {
                printf("%d", matrice[i][j]);
            }
            printf("\n");
        }
    
    // Appel de la fonction pour calculer et afficher la somme des éléments positifs

        sommePositifs(matrice);
    
        return 0;
    }
